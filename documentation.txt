%   -----------                                                                     %
%     - <options> (optional): can have 'firstrow=', 'contrastcol=',                 %
%                             or 'defaultcol='.                                     %
%                             'firstrow', when the row colors start.                %
%                             'contrastcol=', the color of the contrast rows.       %
%                             'defaultcol=', the table's default color.             %
%                             By default, 'firstrow=1' and 'contrastcol=gray!25'.   %
%     - <columns>: The number and type of columns.                                  %
%                  Can be 'c' (centered), 'l' (left-justified),                     %
%                  'r' (right-justified), 'p{<width>}' (top),                       %
%                  'm{<width>}' {middle, vertically} or 'b{<width>}' (bottom).      %
%                  Can also have '|' for a border or '||' for a double border.      %
%                  Can also add '>{\center\arraybackslash}',                        %
%                  '>{\raggedright\arraybackslash}' or                              %
%                  '>{\raggedleft\arraybackslash}' before one of the column types   %
%                  to align in the center, left or right respectively.              %
%     - <label>: label to be able to reference the table.                           %
%     - <caption>: the caption of the table, shown underneath it.                   %
%     - <position> (optional): the position of the figure.                          %
%                              Can be 'h', 'H', 't', 'b' or a combination of them.  %
%                              By default, 'h' is used.                             %
%                                                                                   %
% \myref[]{<label>}                                                                 %
%   Description:                                                                    %
%   ------------                                                                    %
%     Reference a label and adds the type of label as text before the number.       %
%   Parameters:                                                                     %
%   -----------                                                                     %
%     - <prefix> (optional): text to add to the begin of the reference.             %
%                            By default, prefix is ''.                              %
%     - <label>: the label with a prefix.                                           %
%                The prefixes can be 'abst:' for the abstract,                      %
%                'intro:' for the introduction, 'part:' for a part,                 %
%                'chap:' for a chapter, 'sec:' for a section,                       %
%                'subsec:' for a subsection, 'subsubsection:' for a subsubsection,  %
%                'ann:' for an annex, 'toc:' for the table of contents,             %
%                'lof:' for the list of figures, 'lot' for the list of tables,      %
%                'lop:' for the list of programs, 'loe:' for the list of equations, %
%                'fig:' for figures, 'tab:' for tables, 'code:' for programs,       %
%                or 'eq:' for equations.                                            %
%                                                                                   %
% \myseeref{<label>}                                                                %
%   Description:                                                                    %
%   ------------                                                                    %
%     Reference a label and adds 'see' as well as the type of label                 %
%     as text before the number.                                                    %
%   Parameters:                                                                     %
%   -----------                                                                     %
%     - <label>: the label with a prefix.                                           %
%                The prefixes can be 'abst:' for the abstract,                      %
%                'intro:' for the introduction, 'part:' for a part,                 %
%                'chap:' for a chapter, 'sec:' for a section,                       %
%                'subsec:' for a subsection, 'subsubsection:' for a subsubsection,  %
%                'ann:' for an annex, 'toc:' for the table of contents,             %
%                'lof:' for the list of figures, 'lot' for the list of tables,      %
%                'lop:' for the list of programs, 'loe:' for the list of equations, %
%                'fig:' for figures, 'tab:' for tables, 'code:' for programs,       %
%                or 'eq:' for equations.                                            %
%                                                                                   %
