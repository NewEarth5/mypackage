\mysection*{Syntax}
\begin{demosyntax}
\myabstract[<label>]      % label: abst:<label>; added to the TOC
\myabstract*[<label>]     % not added to the TOC
\end{demosyntax}

\myabstract[abs]

page \pageref{abst:abs}.
\end{demo}
